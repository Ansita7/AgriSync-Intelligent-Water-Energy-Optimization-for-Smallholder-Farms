import React, { useState } from 'react';
import { BrowserRouter as Router, Routes, Route } from 'react-router-dom';
import Home from './pages/Home';
import SetupPage from './pages/SetupPage';
import DashboardPage from './pages/DashboardPage';
import './App.css';

function App() {
  const [farm, setFarm] = useState(null);

  return (
    <Router>
      <div className="App">
        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/setup" element={<SetupPage setFarm={setFarm} />} />
          <Route path="/dashboard" element={farm ? <DashboardPage farm={farm} setFarm={setFarm} /> : <Home />} />
        </Routes>
      </div>
    </Router>
  );
}

export default App;

import React, { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import api from '../services/api';
import '../styles/SetupPage.css';

export default function SetupPage({ setFarm }) {
  const navigate = useNavigate();
  const [form, setForm] = useState({farmName:'Sharma Farm',location:'Sangamner, Maharashtra',area:1.5,crop:'wheat',soilType:'loam',irrigationType:'drip',pumpPower:2.2,waterAvailable:1500,solarCapacity:5,electricityCost:5.5});
  
  const handleChange = (e) => {
    const { name, value } = e.target;
    setForm(prev => ({...prev,[name]:isNaN(value)?value:parseFloat(value)}));
  };
  
  const handleSubmit = (e) => {
    e.preventDefault();
    setFarm({id:1,...form,soilMoisture:65,rainProbability:15,createdAt:new Date().toLocaleDateString()});
    navigate('/dashboard');
  };
  
  return (
    <div className="setup-page">
      <div className="setup-container">
        <h1>🌱 Farm Setup</h1>
        <form onSubmit={handleSubmit} className="setup-form">
          <fieldset><legend>Farm Information</legend>
            <div className="form-group"><label>Farm Name</label><input type="text" name="farmName" value={form.farmName} onChange={handleChange} required/></div>
            <div className="form-group"><label>Location</label><input type="text" name="location" value={form.location} onChange={handleChange} required/></div>
            <div className="form-group"><label>Farm Area (hectares)</label><input type="number" name="area" value={form.area} onChange={handleChange} step="0.1" min="0.5" required/></div>
          </fieldset>
          <fieldset><legend>Crop & Soil</legend>
            <div className="form-group"><label>Crop</label><select name="crop" value={form.crop} onChange={handleChange}><option value="wheat">Wheat</option><option value="rice">Rice</option><option value="sugarcane">Sugarcane</option><option value="cotton">Cotton</option></select></div>
            <div className="form-group"><label>Soil</label><select name="soilType" value={form.soilType} onChange={handleChange}><option value="clay">Clay</option><option value="loam">Loam</option><option value="sandy">Sandy</option><option value="silt">Silt</option></select></div>
          </fieldset>
          <fieldset><legend>Irrigation System</legend>
            <div className="form-group"><label>Type</label><select name="irrigationType" value={form.irrigationType} onChange={handleChange}><option value="flood">Flood</option><option value="sprinkler">Sprinkler</option><option value="drip">Drip</option></select></div>

import React, { useState } from 'react';

export default function Optimizer({ farm }) {
  const [water, setWater] = useState(65);
  const [rain, setRain] = useState(15);
  const [solar, setSolar] = useState(35);

  const shouldIrrrigate = water < 70 && rain < 40;
  const needed = shouldIrrigate ? Math.max(0, 1000 - (water * 10)) : 0;
  const cost = (needed / 470) * farm.electricityCost;

  return (
    <div className="component">
      <h2>⚙️ Interactive Optimizer</h2>
      <div className="optimizer-box">
        <div className="slider-group"><label><span>Soil Moisture: {water}%</span><input type="range" min="20" max="100" value={water} onChange={(e) => setWater(Number(e.target.value))} className="slider"/></label></div>
        <div className="slider-group"><label><span>Rain Probability: {rain}%</span><input type="range" min="0" max="100" value={rain} onChange={(e) => setRain(Number(e.target.value))} className="slider"/></label></div>
        <div className="slider-group"><label><span>Solar Available: {solar}%</span><input type="range" min="0" max="100" value={solar} onChange={(e) => setSolar(Number(e.target.value))} className="slider"/></label></div>
      </div>
      <div className={`optimizer-result ${shouldIrrigate ? 'yes' : 'no'}`}>
        <h3>{shouldIrrigate ? '✅ IRRIGATE TODAY' : '⏸️ SKIP TODAY'}</h3>
        {shouldIrrigate && <div className="result-details"><div className="result-item"><span>Water</span><strong>{needed.toFixed(0)}L</strong></div><div className="result-item"><span>Cost</span><strong>₹{cost.toFixed(2)}</strong></div></div>}
      </div>
    </div>
  );
}

import React, { useState } from 'react';

export default function DRInterface({ farm }) {
  const [active, setActive] = useState(true);
  const energy = (farm.pumpPower * 90) / 60;
  const payment = energy * 15;
  const farmer = payment * 0.6;

  return (
    <div className="component">
      <h2>💰 Demand Response</h2>
      {active ? (
        <div className="dr-active">
          <div className="dr-header"><h3>🚨 GRID NEEDS HELP NOW!</h3><p>Pause irrigation 2:00-3:30 PM</p></div>
          <div className="earnings"><div className="earning-item"><span>Your Earnings (60%)</span><strong className="big">₹{farmer.toFixed(2)}</strong></div></div>
          <button className="btn-accept" onClick={() => setActive(false)}>✅ Accept & Pause</button>
        </div>
      ) : (
        <div className="dr-accepted"><h3>✅ Thank You!</h3><p>You earned ₹{farmer.toFixed(2)}</p><button className="btn-reset" onClick={() => setActive(true)}>Next Event</button></div>
      )}
    </div>
  );
}
